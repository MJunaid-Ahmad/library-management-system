import bookModel from "../model/bookModel.js";

async function searchBooks(req, res, next) {
  try {
    let title = req.query.title;
    if (title === undefined || title === "") {
      res.status(400).json({
        success: false,
        message: "Title is required",
      });
    } else {
      let matchedBook = await bookModel.find({
        title: {
          $regex: title,
          $options: "i",
        },
      });
      res.status(200).json({
        success: true,
        Books: matchedBook,
      });
    }
  } catch (err) {
    next(err);
  }
}

export {
  searchBooks
};
